keytool -keystore kafka.server.keystore.jks -alias kafka -validity 365 -genkey -keyalg RSA \
  -storepass 12345678 -keypass 12345678 \
  -dname "CN=kafka1.local, OU=IT, O=Company, L=City, ST=State, C=RU"

keytool -keystore kafka.server.keystore.jks -alias kafka -certreq -file cert-file.csr \
  -storepass 12345678

openssl x509 -req -CA ca-cert.pem -CAkey ca-key.pem -in cert-file.csr -out cert-signed.pem -days 365 -CAcreateserial

# Импортируем CA-сертификат
keytool -keystore kafka.server.keystore.jks -alias CARoot -import -file ca-cert.pem \
  -storepass 12345678 -noprompt

# Импортируем подписанный сертификат брокера
keytool -keystore kafka.server.keystore.jks -alias kafka -import -file cert-signed.pem \
  -storepass 12345678

keytool -keystore kafka.server.truststore.jks -alias CARoot -import -file ca-cert.pem \
  -storepass 12345678 -noprompt


# Для клиента
openssl genpkey -algorithm RSA -out client.key -pkeyopt rsa_keygen_bits:2048




openssl x509 -req -CA ca-cert.pem -CAkey ca-key.pem -in cert-file.csr -out cert-signed.pem -days 365 -CAcreateserial

// broker keystore
keytool -keystore kafka.server.keystore.jks -alias kafka -validity 365 -genkey -keyalg RSA \
  -storepass ksbroker1 -keypass kpbroker1 \
  -dname "CN=broker1.local, OU=IT, O=Company, L=City, ST=State, C=RU"


// подписание и импорт сертов
// cert request
keytool -keystore kafka.server.keystore.jks -alias kafka -certreq -file cert-file -storepass ksbroker1
// create signed cert 
openssl x509 -req -CA ca-cert -CAkey ca-key -in cert-file -out cert-signed -days 365 -CAcreateserial
// import ca
keytool -keystore kafka.server.keystore.jks -alias CARoot -import -file ca-cert -storepass ksbroker1 -noprompt
// import cert
keytool -keystore kafka.server.keystore.jks -alias kafka -import -file cert-signed -storepass ksbroker1


// create client truststore
keytool -keystore kafka.server.truststore.jks -alias CARoot -import -file ca-cert -storepass tsclient -noprompt
