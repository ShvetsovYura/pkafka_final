# для клиента
# Генерация ключа и CSR (Certificate Signing Request)
keytool -keystore client.keystore.jks -alias client -validity 365 -genkey -keyalg RSA \
  -storepass clientpass -keypass clientpass \
  -dname "CN=client1, OU=IT, O=Company, L=City, ST=State, C=RU"

keytool -keystore client.keystore.jks -alias client -certreq -file client-cert-file.csr \
  -storepass clientpass

openssl x509 -req -CA ca-cert.pem -CAkey ca-key.pem -in client-cert-file.csr -out client-cert-signed.pem -days 365 -CAcreateserial

# Импортируем CA сертификат
keytool -keystore client.keystore.jks -alias CARoot -import -file ca-cert.pem \
  -storepass clientpass -noprompt

# Импортируем подписанный клиентский сертификат
keytool -keystore client.keystore.jks -alias client -import -file client-cert-signed.pem \
  -storepass clientpass

# trustore (если нужно)
 keytool -keystore client.truststore.jks -alias CARoot -import -file ca-cert.pem \
  -storepass clienttrustpass -noprompt

  # Извлечение клиентского сертификата и ключа в PEM формате
keytool -importkeystore -srckeystore client.keystore.jks -destkeystore client.p12 -srcstoretype jks -deststoretype pkcs12 -srcstorepass clientpass -deststorepass clientpass
openssl pkcs12 -in client.p12 -out client.pem -nodes -passin pass:clientpass

# Разделение на ключ и сертификат
openssl pkcs12 -in client.p12 -nocerts -out client-key.pem -nodes -passin pass:clientpass
openssl pkcs12 -in client.p12 -nokeys -out client-cert.pem -passin pass:clientpass
