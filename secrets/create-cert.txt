openssl req -new -x509 -keyout ca-key.pem -out ca-cert.pem -days 365 -subj "/CN=Kafka-CA" -nodes

keytool -keystore kafka.server.keystore.jks -alias kafka -validity 365 -genkey -keyalg RSA \
  -storepass 12345678 -keypass 12345678 \
  -dname "CN=kafka1.local, OU=IT, O=Company, L=City, ST=State, C=RU"

keytool -keystore kafka.server.keystore.jks -alias kafka -certreq -file cert-file.csr \
  -storepass 12345678

openssl x509 -req -CA ca-cert.pem -CAkey ca-key.pem -in cert-file.csr -out cert-signed.pem -days 365 -CAcreateserial

# Импортируем CA-сертификат
keytool -keystore kafka.server.keystore.jks -alias CARoot -import -file ca-cert.pem \
  -storepass 12345678 -noprompt

# Импортируем подписанный сертификат брокера
keytool -keystore kafka.server.keystore.jks -alias kafka -import -file cert-signed.pem \
  -storepass 12345678

keytool -keystore kafka.server.truststore.jks -alias CARoot -import -file ca-cert.pem \
  -storepass 12345678 -noprompt
