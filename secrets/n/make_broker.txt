# 1. Создаем приватный ключ (без пароля)
openssl genpkey -algorithm RSA -out client.key -pkeyopt rsa_keygen_bits:2048

# 2. Создаем CSR (Certificate Signing Request)
openssl req -new -key client.key -out client.csr -subj "/CN=kafka-client/O=MyCompany"

# 3. Подписываем CSR корневым CA (из предыдущих шагов)
openssl x509 -req -in client.csr -CA ../ca.crt -CAkey ../ca.key -CAcreateserial -out client.pem -days 365 -sha256

# 4. Проверяем сертификат
openssl x509 -in client.pem -text -noout
